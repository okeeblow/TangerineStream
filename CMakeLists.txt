cmake_minimum_required(VERSION 3.20)
project(ml_tangerinestream VERSION 0.2.0 LANGUAGES CXX RC)

set(WINAMP_SDK "${CMAKE_CURRENT_SOURCE_DIR}/WA5.55_SDK" CACHE PATH "Nullsoft Winamp SDK")
if(NOT EXISTS "${WINAMP_SDK}/gen_ml/ml.h")
    message(FATAL_ERROR "Set WINAMP_SDK to the extracted Nullsoft Winamp SDK")
endif()

add_library(ml_tangerinestream SHARED
    src/plugin.cpp src/server.cpp src/store.cpp src/playback.cpp src/artwork.cpp src/plugin.rc)
set_target_properties(ml_tangerinestream PROPERTIES PREFIX "")
target_compile_features(ml_tangerinestream PRIVATE cxx_std_17)
target_compile_definitions(ml_tangerinestream PRIVATE
    UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_include_directories(ml_tangerinestream PRIVATE "${WINAMP_SDK}")
target_link_libraries(ml_tangerinestream PRIVATE
    comctl32 winhttp ws2_32 crypt32 bcrypt windowscodecs ole32 gdi32 uuid)

if(MINGW)
    if(CMAKE_SIZEOF_VOID_P EQUAL 4)
        set(sqlite_machine i386)
        set(sqlite_def src/winsqlite3-x86.def)
        set(sqlite_options --kill-at)
    else()
        set(sqlite_machine i386:x86-64)
        set(sqlite_def src/winsqlite3.def)
    endif()
    set(sqlite_import "${CMAKE_CURRENT_BINARY_DIR}/libwinsqlite3.a")
    add_custom_command(OUTPUT "${sqlite_import}"
        COMMAND "${CMAKE_DLLTOOL}" -m "${sqlite_machine}" ${sqlite_options}
            -d "${CMAKE_CURRENT_SOURCE_DIR}/${sqlite_def}" -l "${sqlite_import}"
        DEPENDS "${sqlite_def}" VERBATIM)
    add_custom_target(sqlite_import DEPENDS "${sqlite_import}")
    add_dependencies(ml_tangerinestream sqlite_import)
    target_link_libraries(ml_tangerinestream PRIVATE "${sqlite_import}")
    target_link_options(ml_tangerinestream PRIVATE -static -static-libgcc -static-libstdc++)
    target_compile_options(ml_tangerinestream PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>")
else()
    set_property(TARGET ml_tangerinestream PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_link_libraries(ml_tangerinestream PRIVATE winsqlite3)
endif()

set(WACUP_PLUGINS_DIR "" CACHE PATH "WACUP Plugins directory (empty to detect the installed copy)")
set(plugin_destination "${WACUP_PLUGINS_DIR}")
if(NOT plugin_destination)
    set(plugin_destination "$ENV{WACUP_PLUGINS_DIR}")
endif()

if(NOT plugin_destination AND CMAKE_HOST_WIN32 AND CMAKE_VERSION VERSION_GREATER_EQUAL 3.24)
    foreach(hive IN ITEMS HKCU HKLM)
        set(uninstall_key "${hive}/SOFTWARE/Microsoft/Windows/CurrentVersion/Uninstall/WACUP")
        cmake_host_system_information(RESULT wacup_dir QUERY WINDOWS_REGISTRY
            "${uninstall_key}" VALUE InstallLocation VIEW 32)
        if(NOT wacup_dir OR NOT IS_DIRECTORY "${wacup_dir}/Plugins")
            cmake_host_system_information(RESULT wacup_icon QUERY WINDOWS_REGISTRY
                "${uninstall_key}" VALUE DisplayIcon VIEW 32)
            string(REGEX REPLACE ",[ \t]*-?[0-9]+$" "" wacup_exe "${wacup_icon}")
            string(REPLACE "\"" "" wacup_exe "${wacup_exe}")
            get_filename_component(wacup_dir "${wacup_exe}" DIRECTORY)
        endif()
        if(wacup_dir AND IS_DIRECTORY "${wacup_dir}/Plugins")
            set(plugin_destination "${wacup_dir}/Plugins")
            break()
        endif()
    endforeach()
endif()

if(NOT plugin_destination)
    foreach(program_files IN ITEMS "$ENV{ProgramFiles\(x86\)}" "$ENV{ProgramFiles}")
        if(program_files AND IS_DIRECTORY "${program_files}/WACUP/Plugins")
            set(plugin_destination "${program_files}/WACUP/Plugins")
            break()
        endif()
    endforeach()
endif()

if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    install(CODE [[message(FATAL_ERROR "Use a 32-bit build to install into WACUP.")]])
elseif(NOT plugin_destination)
    install(CODE [[message(FATAL_ERROR "WACUP was not found. Set WACUP_PLUGINS_DIR to its Plugins directory.")]])
else()
    file(TO_CMAKE_PATH "${plugin_destination}" plugin_destination)
    cmake_path(ABSOLUTE_PATH plugin_destination BASE_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" NORMALIZE)
    message(STATUS "WACUP plugin install destination: ${plugin_destination}")
    install(TARGETS ml_tangerinestream RUNTIME DESTINATION "${plugin_destination}")
endif()
